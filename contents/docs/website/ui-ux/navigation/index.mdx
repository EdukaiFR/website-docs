# Navigation

Edukai implements a comprehensive navigation system with a responsive sidebar, breadcrumbs, and contextual navigation patterns to provide intuitive user experience across the platform.

## 🗂️ Navigation Architecture

### Route Structure
The navigation follows Next.js App Router patterns with grouped routes:

```
app/
├── (auth)/          # Authentication routes
│   ├── auth/
│   ├── auth-demo/
│   └── change-password/
└── (dashboard)/     # Main application routes
    ├── club-edukai/
    ├── generate/
    ├── library/
    ├── notifications/
    ├── settings/
    └── support/
```

## 🍞 Breadcrumb Navigation

### Implementation
```tsx
// components/ui/breadcrumb.tsx
import {
  Breadcrumb,
  BreadcrumbItem,
  BreadcrumbLink,
  BreadcrumbList,
  BreadcrumbPage,
  BreadcrumbSeparator,
} from "@/components/ui/breadcrumb"

function PageBreadcrumb({ items }: { items: BreadcrumbItem[] }) {
  return (
    <Breadcrumb>
      <BreadcrumbList>
        {items.map((item, index) => (
          <React.Fragment key={item.href}>
            <BreadcrumbItem>
              {index === items.length - 1 ? (
                <BreadcrumbPage>{item.label}</BreadcrumbPage>
              ) : (
                <BreadcrumbLink href={item.href}>
                  {item.label}
                </BreadcrumbLink>
              )}
            </BreadcrumbItem>
            {index < items.length - 1 && <BreadcrumbSeparator />}
          </React.Fragment>
        ))}
      </BreadcrumbList>
    </Breadcrumb>
  )
}
```

### Active Link Detection
```tsx
function NavLink({ href, children }: { href: string; children: React.ReactNode }) {
  const pathname = usePathname()
  const isActive = pathname === href || pathname.startsWith(href + '/')

  return (
    <Link
      href={href}
      className={cn(
        "flex items-center px-3 py-2 rounded-md text-sm font-medium transition-colors",
        isActive
          ? "bg-sidebar-accent text-sidebar-accent-foreground"
          : "text-sidebar-foreground hover:bg-sidebar-accent hover:text-sidebar-accent-foreground"
      )}
    >
      {children}
    </Link>
  )
}
```

## 📋 Contextual Navigation

### Course Library Navigation
```tsx
// library/[id]/sections/ navigation
const courseSections = [
  { id: "overview", label: "Overview", icon: "📊" },
  { id: "quiz", label: "Quizzes", icon: "❓" },
  { id: "exams", label: "Exams", icon: "📝" },
  { id: "resume-files", label: "Summary", icon: "📄" },
  { id: "objectives", label: "Objectives", icon: "🎯" },
  { id: "statistics", label: "Statistics", icon: "📈" },
  { id: "similar-courses", label: "Similar", icon: "🔍" }
]

function CourseSectionNav({ activeSection }: { activeSection: string }) {
  return (
    <nav className="flex space-x-1 border-b">
      {courseSections.map((section) => (
        <Link
          key={section.id}
          href={`/library/${courseId}/sections/${section.id}`}
          className={cn(
            "flex items-center px-3 py-2 text-sm font-medium border-b-2 transition-colors",
            activeSection === section.id
              ? "border-primary text-primary"
              : "border-transparent text-muted-foreground hover:text-foreground"
          )}
        >
          <span className="mr-2">{section.icon}</span>
          {section.label}
        </Link>
      ))}
    </nav>
  )
}
```

### Step-by-Step Navigation
```tsx
// Generate workflow navigation
function GenerateStepNav({ currentStep }: { currentStep: number }) {
  const steps = [
    { id: 1, label: "Upload", description: "Upload your files" },
    { id: 2, label: "Process", description: "Extract content" },
    { id: 3, label: "Generate", description: "Create quiz" },
    { id: 4, label: "Review", description: "Review & edit" }
  ]

  return (
    <nav className="flex justify-between">
      {steps.map((step, index) => (
        <div
          key={step.id}
          className={cn(
            "flex flex-col items-center",
            currentStep >= step.id ? "text-primary" : "text-muted-foreground"
          )}
        >
          <div className={cn(
            "w-8 h-8 rounded-full flex items-center justify-center border-2 mb-2",
            currentStep >= step.id
              ? "border-primary bg-primary text-primary-foreground"
              : "border-muted"
          )}>
            {step.id}
          </div>
          <span className="text-sm font-medium">{step.label}</span>
          <span className="text-xs text-muted-foreground">{step.description}</span>
        </div>
      ))}
    </nav>
  )
}
```
