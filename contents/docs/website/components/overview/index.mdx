# Components Structure - Edukai Frontend 📦

## Overview

The components directory is organized by feature and purpose, following a hierarchical structure that promotes reusability and maintainability. Components are built using TypeScript, Tailwind CSS, and ShadCN UI primitives.

## Component Directory Structure

```
📁 components/
├── 📁 auth/                      # Authentication components
├── 📁 badge/                     # Badge/status indicator components
├── 📁 card/                      # Card components
├── 📁 data-table/               # Data table components
├── 📁 generator/                # Content generation UI
├── 📁 input/                    # Input components
├── 📁 recognition/              # OCR/text recognition components
├── 📁 settings/                 # Settings-related components
├── 📁 sidebar/                  # Sidebar navigation components
├── 📁 ui/                       # ShadCN UI base components (30+ components)
├── 📄 mode-toggle.tsx           # Dark/light mode toggle
├── 📄 navigation-location.tsx   # Navigation breadcrumb (284 lines)
└── 📄 theme-provider.tsx        # Theme context provider
```

  children: React.ReactNode;
  defaultTheme?: "dark" | "light" | "system";
  storageKey?: string;
}

export function ThemeProvider({
  children,
  defaultTheme = "system",
  storageKey = "theme",
}: ThemeProviderProps) {
  const [theme, setTheme] = useState<"dark" | "light" | "system">(defaultTheme);

  return (
    <ThemeProviderContext.Provider value={{ theme, setTheme }}>
      {children}
    </ThemeProviderContext.Provider>
  );

}

export function DataFetcher<T>({ url, children }: DataFetcherProps<T>) {

  const [error, setError] = useState<string | null>(null);

// Usage

### 3. **Higher-Order Components**

```typescript
// Authentication HOC
export function withAuth<P extends object>(Component: React.ComponentType<P>) {
  return function AuthenticatedComponent(props: P) {
    const { user, loading } = useSession();

    if (loading) return <LoadingSkeleton />;
    if (!user) return <Navigate to="/auth/login" />;

    return <Component {...props} />;
  };
}

// Usage
const ProtectedDashboard = withAuth(Dashboard);
```

### 2. **Memoization**

```typescript
// Memoize expensive components
export const ExpensiveComponent = memo(({ data, onAction }) => {
  const processedData = useMemo(
    () => data.map((item) => ({ ...item, processed: true })),
    [data]
  );

  const handleAction = useCallback(
    (id: string) => {
      onAction(id);
    },
    [onAction]
  );

  return (
    <div>
      {processedData.map((item) => (
        <Item key={item.id} data={item} onAction={handleAction} />
      ))}
    </div>
  );
});
```
