# Authentication Services 🔗

## Overview

The **Edukai Authentication Services** provide a comprehensive API service layer for frontend-backend communication. The service layer handles all authentication operations including login, registration, logout, and token management with proper error handling and TypeScript safety.

## Core Auth Service

### 🔗 **useAuthService Hook**

**File**: `services/auth.ts` (100 lines)

The main authentication service that handles all API communication with the backend.

#### **Service Interface**

```typescript
export interface AuthService {
  login: (credentials: LoginCredentials) => Promise<AuthResponse>;
  register: (userData: RegisterData) => Promise<AuthResponse>;
  logout: () => Promise<void>;
  refreshToken: () => Promise<{ token: string }>;
}

export function useAuthService(): AuthService {
  const apiUrl = process.env.NEXT_PUBLIC_API_URL;

  if (!apiUrl) {
    console.error(
      "❌ NEXT_PUBLIC_API_URL n'est pas défini ! Veuillez l'ajouter à votre fichier .env.local"
    );
  }

  return {
    login,
    register,
    logout,
    refreshToken,
  };
}
```

#### **Type Definitions**

```typescript
// Login credentials interface
export interface LoginCredentials {
  username: string;
  password: string;
}

// Registration data interface
export interface RegisterData {
  username: string;
  password: string;
  email?: string;
  firstName?: string;
  lastName?: string;
}

// Authentication response interface
export interface AuthResponse {
  token: string;
  user: {
    id: string;
    username: string;
    email?: string;
    firstName?: string;
    lastName?: string;
  };
}
```

#### **Login Method**

```typescript
const login = async (credentials: LoginCredentials): Promise<AuthResponse> => {
  try {
    const response = await axios.post(`${apiUrl}/auth/login`, credentials, {
      withCredentials: true, // Include cookies for CSRF protection
    });
    return response.data;
  } catch (error: any) {
    console.error(
      "Erreur de connexion:",
      error.response?.data?.message || error.message
    );
    throw error;
  }
};
```

**Features**:

- **CSRF Protection**: `withCredentials: true` for cookie-based CSRF tokens
- **Error Handling**: Comprehensive error logging and re-throwing
- **Type Safety**: Full TypeScript interface compliance
- **Environment Configuration**: Dynamic API URL from environment variables

#### **Signin Action**

```typescript
export async function signinAction(
  credentials: SigninFormValues
): Promise<AuthResponse> {
  const authService = useAuthService();

  try {
    console.log("🔐 [Auth Action] Attempting signin for:", credentials.email);

    // Convert form values to service format
    const loginCredentials: LoginCredentials = {
      username: credentials.email,
      password: credentials.password,
    };

    const response = await authService.login(loginCredentials);

    // Store session data
    sessionStorage.setToken(response.token);
    sessionStorage.setUser(response.user);

    console.log("✅ [Auth Action] Signin successful");
    return { success: true, data: response };
  } catch (error: any) {
    console.error("❌ [Auth Action] Signin failed:", error.message);
    const errorMessage = translateApiError(
      error.response?.data?.message || error.message
    );
    return { success: false, error: errorMessage };
  }
}
```

**Features**:

- **Form Integration**: Converts form values to service format
- **Session Management**: Automatic token and user storage
- **Error Translation**: French localized error messages
- **Logging**: Comprehensive debug logging with emojis
- **Type Safety**: Full TypeScript compliance

 * Get authorization headers for API calls

export function getAuthHeaders() {

#### **User Management**

```typescript
/**
 * Get the current authenticated user from session storage
 * @returns User object or null if not authenticated
 */
export function getCurrentUser() {
  return sessionStorage.getUser();
}

/**
 * Get the current user's ID from session storage
 * @returns User ID string or null if not authenticated
 */
export function getCurrentUserId(): string | null {
  const user = getCurrentUser();
  console.log("🔍 [Auth Utils] Debug - Full user object:", user);

  if (!user) {
    return null;
  }

  const userId = user.id || user._id || user.userId;
  console.log("🔍 [Auth Utils] Debug - Extracted userId:", userId);

  return userId || null;
}
```

#### **Authentication State**

```typescript
/**
 * Check if user is authenticated
 * @returns boolean indicating if user is logged in
 */
export function isAuthenticated(): boolean {
  const token = sessionStorage.getToken();
  const user = sessionStorage.getUser();
  return !!(token && user);
}
```

export function translateApiError(error: string): string {
  const translations = {
    // Authentication errors
    "Invalid credentials": "Identifiants invalides",

    "Authentication failed": "Échec de l'authentification",

    "Password too weak": "Mot de passe trop faible",
    "Invalid email format": "Format d'email invalide",

    // Network errors
    "Network Error": "Erreur de connexion réseau",
    "Request timeout": "Délai d'attente dépassé",
    "Server error": "Erreur serveur",

    // Session errors
    "Session expired": "Session expirée",
    "Token invalid": "Token invalide",
    Unauthorized: "Non autorisé",

  return translations[error] || error;
}

## Service Integration

### 🔌 **Hook Integration**

#### **useSession Hook Integration**

**File**: `hooks/useSession.ts` (117 lines)

The useSession hook integrates the auth service with React state management:

```typescript
export function useSession() {
  const [user, setUser] = useState<User>(null);
  const [loading, setLoading] = useState(true);
  const authService = useAuthService();

  const login = async (credentials: LoginCredentials) => {
    try {
      const response = await authService.login(credentials);
      sessionStorage.setToken(response.token);
      sessionStorage.setUser(response.user);
      setUser(response.user);
      authToast.loginSuccess();
      return { success: true, data: response };
    } catch (error: any) {
      const errorMessage = translateApiError(
        error.response?.data?.message || error.message
      );
      authToast.loginError(errorMessage);
      return { success: false, error: errorMessage };
    }
  };

  const register = async (userData: RegisterData) => {
    try {
      const response = await authService.register(userData);
      sessionStorage.setToken(response.token);
      sessionStorage.setUser(response.user);
      setUser(response.user);
      authToast.registerSuccess();
      return { success: true, data: response };
    } catch (error: any) {
      const errorMessage = translateApiError(
        error.response?.data?.message || error.message
      );
      authToast.registerError(errorMessage);
      return { success: false, error: errorMessage };
    }
  };

  const logout = async () => {
    try {
      await authService.logout();
      sessionStorage.clearSession();
      setUser(null);
      authToast.logoutSuccess();
    } catch (error) {
      console.error("Erreur lors de la déconnexion:", error);
      authToast.logoutError();
      // Clear local session even if server logout fails
      sessionStorage.clearSession();
      setUser(null);
    }
  };

  return { user, loading, login, register, logout };
}
```

## Performance Considerations

### ⚡ **Service Optimizations**

#### **1. Request Caching**

```typescript
// Cache token refresh requests
let refreshPromise: Promise<{ token: string }> | null = null;

const refreshToken = async (): Promise<{ token: string }> => {
  if (refreshPromise) {
    return refreshPromise;
  }

  refreshPromise = axios
    .post(
      `${apiUrl}/auth/refresh`,
      {},
      {
        withCredentials: true,
      }
    )
    .then((response) => response.data);

  try {
    const result = await refreshPromise;
    refreshPromise = null;
    return result;
  } catch (error) {
    refreshPromise = null;
    throw error;
  }
};
```

#### **2. Request Deduplication**

```typescript
// Deduplicate concurrent login requests
const pendingRequests = new Map();

const login = async (credentials: LoginCredentials) => {
  const key = `${credentials.username}:${credentials.password}`;

  if (pendingRequests.has(key)) {
    return pendingRequests.get(key);
  }

  const promise = axios
    .post(`${apiUrl}/auth/login`, credentials, {
      withCredentials: true,
    })
    .then((response) => response.data);

  pendingRequests.set(key, promise);

  try {
    const result = await promise;
    pendingRequests.delete(key);
    return result;
  } catch (error) {
    pendingRequests.delete(key);
    throw error;
  }
};
```

#### **3. Error Recovery**

```typescript
// Automatic retry for network errors
const retryRequest = async (fn: () => Promise<any>, maxRetries = 3) => {
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catch (error: any) {
      if (i === maxRetries - 1 || !isRetryableError(error)) {
        throw error;
      }
      await delay(1000 * Math.pow(2, i)); // Exponential backoff
    }
  }
};

const isRetryableError = (error: any) => {
  return error.code === "NETWORK_ERROR" || error.response?.status >= 500;
};
```
