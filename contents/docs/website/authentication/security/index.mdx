```typescript
export function useAuthService(): AuthService {
  const apiUrl = process.env.NEXT_PUBLIC_API_URL;

  // Validate API URL is HTTPS in production
  if (
    process.env.NODE_ENV === "production" &&
    !apiUrl?.startsWith("https://")
  ) {
    console.error("❌ API URL must use HTTPS in production");
    throw new Error("Insecure API configuration");
  }

  const secureAxiosConfig = {
    withCredentials: true, // Include cookies for CSRF protection
    timeout: 10000, // Prevent hanging requests
    headers: {
      "Content-Type": "application/json",
      "X-Requested-With": "XMLHttpRequest", // CSRF protection
    },
  };

  const login = async (
    credentials: LoginCredentials
  ): Promise<AuthResponse> => {
    try {
      const response = await axios.post(
        `${apiUrl}/auth/login`,
        credentials,
        secureAxiosConfig
      );
      return response.data;
    } catch (error: any) {
      // Secure error handling - don't expose internal details
      console.error("Login error:", error.response?.status);
      throw new Error(getSecureErrorMessage(error));
    }
  };
}
```

```typescript
// Client-side rate limiting
class RateLimiter {
  private requests: Map<string, number[]> = new Map();
  private readonly maxRequests = 5;
  private readonly windowMs = 60000; // 1 minute

  canMakeRequest(endpoint: string): boolean {
    const now = Date.now();
    const requests = this.requests.get(endpoint) || [];

    // Remove old requests outside the window
    const validRequests = requests.filter((time) => now - time < this.windowMs);

    if (validRequests.length >= this.maxRequests) {
      return false;
    }

    validRequests.push(now);
    this.requests.set(endpoint, validRequests);
    return true;
  }
}
```

### 💾 **Secure Session Storage**

#### **Token Security**

**File**: `lib/session.ts` (65 lines)

Secure token storage with validation and cleanup.

```typescript
const TOKEN_KEY = "auth_token";
const USER_KEY = "user_data";

export const sessionStorage = {
  setToken: (token: string) => {
    if (typeof window !== "undefined") {
      // Validate token format before storing
      if (!isValidJWTFormat(token)) {
        console.error("Invalid token format");
        return;
      }

      localStorage.setItem(TOKEN_KEY, token);
    }
  },

  getToken: () => {
    if (typeof window !== "undefined") {
      const token = localStorage.getItem(TOKEN_KEY);

      // Validate token before returning
      if (token && token !== "undefined" && token !== "null") {
        if (isValidJWTFormat(token) && !isTokenExpired(token)) {
          return token;
        } else {
          // Remove invalid/expired token
          localStorage.removeItem(TOKEN_KEY);
        }
      }
    }
    return null;
  },
};

// Token validation utilities
function isValidJWTFormat(token: string): boolean {
  const parts = token.split(".");
  return parts.length === 3 && parts.every((part) => part.length > 0);
}

function isTokenExpired(token: string): boolean {
  try {
    const payload = JSON.parse(atob(token.split(".")[1]));
    const exp = payload.exp * 1000; // Convert to milliseconds
    return Date.now() >= exp;
  } catch (error) {
    return true; // Treat invalid tokens as expired
  }
}
```

#### **Data Sanitization**

```typescript
// Secure user data storage with sanitization
setUser: (user: any) => {
  if (typeof window !== "undefined") {
    // Sanitize user data before storage
    const sanitizedUser = sanitizeUserData(user);
    localStorage.setItem(USER_KEY, JSON.stringify(sanitizedUser));
  }
},

getUser: () => {
  if (typeof window !== "undefined") {
    const user = localStorage.getItem(USER_KEY);
    if (user && user !== "undefined") {
      try {
        const parsedUser = JSON.parse(user);
        return sanitizeUserData(parsedUser);
      } catch (error) {
        console.error("Error parsing user data:", error);
        localStorage.removeItem(USER_KEY);
        return null;
      }
    }
  }
  return null;
},

// User data sanitization
function sanitizeUserData(user: any): any {
  if (!user || typeof user !== 'object') return null;

  return {
    id: sanitizeString(user.id),
    username: sanitizeString(user.username),
    email: sanitizeEmail(user.email),
    firstName: sanitizeString(user.firstName),
    lastName: sanitizeString(user.lastName),
    // Only include safe, expected fields
  };
}

function sanitizeString(value: any): string | undefined {
  if (typeof value !== 'string') return undefined;
  return value.trim().substring(0, 255); // Limit length
}

function sanitizeEmail(value: any): string | undefined {
  if (typeof value !== 'string') return undefined;
  const email = value.toLowerCase().trim();
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) ? email : undefined;
}
```

#### **Automatic Cleanup**

```typescript
// Secure cleanup of invalid data
cleanupInvalidData: () => {
  if (typeof window !== "undefined") {
    const token = localStorage.getItem(TOKEN_KEY);
    const user = localStorage.getItem(USER_KEY);

    // Remove invalid token values
    if (token === "undefined" || token === "null" || !isValidJWTFormat(token)) {
      localStorage.removeItem(TOKEN_KEY);
    }

    // Remove invalid user data
    if (user === "undefined" || user === "null") {
      localStorage.removeItem(USER_KEY);
    } else if (user) {
      try {
        const parsedUser = JSON.parse(user);
        if (!isValidUserData(parsedUser)) {
          localStorage.removeItem(USER_KEY);
        }
      } catch (error) {
        localStorage.removeItem(USER_KEY);
      }
    }
  }
},
  function isValidUserData(user: any): boolean {
    return (
      user &&
      typeof user === "object" &&
      typeof user.id === "string" &&
      user.id.length > 0 &&
      user.id.length < 255
    );
```

```typescript
// Secure redirect handling
export function useAuthGuard(options: UseAuthGuardOptions = {}) {
  const { redirectTo = "/auth", enabled = true } = options;
  const { user, loading } = useSession();
  const router = useRouter();

  useEffect(() => {
    if (enabled && !loading && !user) {
      // Security: Validate and sanitize redirect URL
      const currentPath = window.location.pathname + window.location.search;
      const safeRedirectPath = sanitizeRedirectPath(currentPath);
      const redirectUrl = `${redirectTo}?redirect=${encodeURIComponent(
        safeRedirectPath
      )}`;

      router.push(redirectUrl);
    }
  }, [user, loading, router, enabled, redirectTo]);
}

function sanitizeRedirectPath(path: string): string {
  // Security: Only allow safe internal paths
  const allowedPaths = [
    "/dashboard",
    "/profile",
    "/courses",
    "/quiz",
    "/settings",
  ];

  // Remove any dangerous characters
  const sanitized = path.replace(/[^a-zA-Z0-9\/\-_?&=]/g, "");

  // Check if path starts with allowed prefix
  const isAllowed = allowedPaths.some((allowed) =>
    sanitized.startsWith(allowed)
  );

  return isAllowed ? sanitized : "/dashboard";
}
```

```typescript
// Secure token refresh with validation
const validateSession = async () => {
  try {
    // Security: Validate current token before refresh
    const currentToken = sessionStorage.getToken();
    if (!currentToken || !isValidJWTFormat(currentToken)) {
      throw new Error("Invalid current token");
    }

    const response = await authService.refreshToken();

    // Security: Validate new token before storing
    if (!response.token || !isValidJWTFormat(response.token)) {
      throw new Error("Invalid refresh token received");
    }

    sessionStorage.setToken(response.token);
    setLoading(false);
  } catch (error: any) {
    console.warn("Token validation failed, clearing session:", error.message);
    sessionStorage.clearSession();
    setUser(null);
    setLoading(false);
  }
};
```

```typescript
// Regular session integrity validation
useEffect(() => {
  const validateSessionIntegrity = () => {
    const token = sessionStorage.getToken();
    const user = sessionStorage.getUser();

    // Security: Ensure token and user are consistent
    if ((token && !user) || (!token && user)) {
      console.warn("Session integrity violation detected");

      setUser(null);

      setUser(null);

  };
```
