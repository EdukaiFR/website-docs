- Visual regression testing
- Performance testing

## 🧪 Component Testing

### Basic Component Test

```typescript
// __tests__/components/Button.test.tsx
import { render, screen, fireEvent } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { Button } from '@/components/ui/button'

describe('Button Component', () => {
  it('renders with correct text', () => {
    render(<Button>Click me</Button>)
    expect(screen.getByRole('button', { name: /click me/i })).toBeInTheDocument()
  })

  it('handles click events', async () => {
    const handleClick = jest.fn()
    const user = userEvent.setup()
    
    render(<Button onClick={handleClick}>Click me</Button>)
    
    await user.click(screen.getByRole('button'))
    expect(handleClick).toHaveBeenCalledTimes(1)
  })

  it('applies variant classes correctly', () => {
    render(<Button variant="destructive">Delete</Button>)
    const button = screen.getByRole('button')
    expect(button).toHaveClass('bg-destructive')
  })

  it('is disabled when loading', () => {
    render(<Button disabled>Loading...</Button>)
    expect(screen.getByRole('button')).toBeDisabled()
  })
})
```

import { render, screen, waitFor } from '@testing-library/react'

## 🪝 Hook Testing

### Custom Hook Testing

```typescript
// __tests__/hooks/useCourse.test.ts
import { renderHook, act } from '@testing-library/react'
import { useCourse } from '@/hooks/useCourse'

const mockCourseService = {
  createCourse: jest.fn(),
  getCourseById: jest.fn(),
  getCourses: jest.fn(),
  addQuizToCourse: jest.fn(),
  addSheetToCourse: jest.fn(),
}

describe('useCourse Hook', () => {
  beforeEach(() => {
    jest.clearAllMocks()
  })

  it('creates a course successfully', async () => {
    const mockCourse = { id: '123', title: 'Test Course' }
    mockCourseService.createCourse.mockResolvedValue(mockCourse)

    const { result } = renderHook(() => useCourse(mockCourseService))

    await act(async () => {
      const courseId = await result.current.createCourse({
        title: 'Test Course',
        subject: 'math',
        level: 'beginner',
      })
      expect(courseId).toBe('123')
    })

    expect(result.current.courseId).toBe('123')
    expect(mockCourseService.createCourse).toHaveBeenCalledWith(
      'Test Course',
      'math',
      'beginner'
    )
  })

  it('handles creation errors', async () => {
    mockCourseService.createCourse.mockRejectedValue(new Error('Failed'))

    const { result } = renderHook(() => useCourse(mockCourseService))

    await act(async () => {
      try {
        await result.current.createCourse({
          title: 'Test Course',
          subject: 'math',
          level: 'beginner',
        })
      } catch (error) {
        expect(error).toBeInstanceOf(Error)
      }
    })

    expect(result.current.courseError).toBe('Failed to create course. Please try again.')
  })
})
```

### State Management Testing

```typescript
// __tests__/hooks/useQuizState.test.ts
import { renderHook, act } from '@testing-library/react'
import { useQuizState } from '@/hooks/useQuizState'

describe('useQuizState Hook', () => {
  it('starts quiz with correct initial state', () => {
    const { result } = renderHook(() => useQuizState())

    act(() => {
      result.current.dispatch({
        type: 'START_QUIZ',
        payload: { timeLimit: 1800 }, // 30 minutes
      })
    })

    expect(result.current.state).toEqual({
      currentQuestion: 0,
      answers: {},
      timeRemaining: 1800,
      status: 'active',
      score: null,
    })
  })

  it('records answers correctly', () => {
    const { result } = renderHook(() => useQuizState())

    act(() => {
      result.current.dispatch({
        type: 'START_QUIZ',
        payload: { timeLimit: 1800 },
      })
    })

    act(() => {
      result.current.dispatch({
        type: 'ANSWER_QUESTION',
        payload: { questionIndex: 0, answer: 'Paris' },
      })
    })

    expect(result.current.state.answers[0]).toBe('Paris')
  })

  it('completes quiz when time runs out', () => {
    const { result } = renderHook(() => useQuizState())

    act(() => {
      result.current.dispatch({
        type: 'START_QUIZ',
        payload: { timeLimit: 1 },
      })
    })

    act(() => {
      result.current.dispatch({ type: 'TICK_TIMER' })
    })

    expect(result.current.state.status).toBe('completed')
    expect(result.current.state.timeRemaining).toBe(0)
  })
})
```

import { createMocks } from 'node-mocks-http'
import handler from '@/pages/api/courses'

describe('/api/courses', () => {

### Custom Cypress Commands

```typescript
// cypress/support/commands.ts
declare global {
  namespace Cypress {
    interface Chainable {
      login(email: string, password: string): Chainable<void>
      createCourse(courseData: any): Chainable<void>
      uploadFile(filePath: string): Chainable<void>
    }
  }
}

Cypress.Commands.add('login', (email: string, password: string) => {
  cy.session([email, password], () => {
    cy.visit('/auth')
    cy.get('[name="email"]').type(email)
    cy.get('[name="password"]').type(password)
    cy.get('[type="submit"]').click()
    cy.url().should('not.include', '/auth')
  })
})

Cypress.Commands.add('createCourse', (courseData) => {
  cy.request('POST', '/api/courses', courseData)
})

Cypress.Commands.add('uploadFile', (filePath) => {
  cy.get('[data-testid="file-input"]').selectFile(filePath)
  cy.get('[data-testid="upload-btn"]').click()
})
```

## 🎯 Best Practices

1. **Test Behavior, Not Implementation**: Focus on what the user sees and does
2. **Use Data Test IDs**: Add `data-testid` attributes for reliable element selection
3. **Mock External Dependencies**: Mock APIs, services, and third-party libraries
4. **Test Edge Cases**: Include error states, empty states, and boundary conditions
5. **Maintain Test Isolation**: Each test should be independent and repeatable
6. **Use Page Object Model**: Organize E2E tests with reusable page objects
7. **Coverage Goals**: Aim for high coverage but focus on critical paths
8. **Performance Testing**: Include performance benchmarks for critical components

Testing ensures Edukai's reliability, maintainability, and user experience quality across all features and user workflows.
