### Key Settings Explained

- **`strict: true`**: Enables all strict type checking options
- **`noEmit: true`**: TypeScript only for type checking, Next.js handles compilation
- **`moduleResolution: "bundler"`**: Modern module resolution for bundlers
- **`paths`**: Path mapping for cleaner imports (`@/` prefix)

## 🔧 Generic Programming

### Generic Hooks

```typescript
// Generic data fetching hook
function useApiData<T>(url: string): {
  data: T | null;
  loading: boolean;
  error: Error | null;
  refetch: () => void;
} {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  const fetchData = useCallback(async () => {
    try {
      setLoading(true);
      setError(null);
      const response = await fetch(url);
      
      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }
      
      const result: T = await response.json();
      setData(result);
    } catch (err) {
      setError(err as Error);
    } finally {
      setLoading(false);
    }
  }, [url]);

  useEffect(() => {
    fetchData();
  }, [fetchData]);

  return { data, loading, error, refetch: fetchData };
}

// Usage with specific types
const { data: courses } = useApiData<Course[]>('/api/courses');
const { data: quiz } = useApiData<Quiz>(`/api/quiz/${quizId}`);
```

## 🛡️ Type Safety Patterns

### Discriminated Unions

```typescript
// State management with discriminated unions
type LoadingState = 
  | { status: 'idle' }
  | { status: 'loading' }
  | { status: 'success'; data: any }
  | { status: 'error'; error: string };

function useAsyncOperation() {
  const [state, setState] = useState<LoadingState>({ status: 'idle' });

  const execute = async (operation: () => Promise<any>) => {
    setState({ status: 'loading' });
    
    try {
      const data = await operation();
      setState({ status: 'success', data });
    } catch (error) {
      setState({ 
        status: 'error', 
        error: error instanceof Error ? error.message : 'Unknown error' 
      });
    }
  };

  return { state, execute };
}

// Type-safe state handling
function MyComponent() {
  const { state, execute } = useAsyncOperation();

  switch (state.status) {
    case 'idle':
      return <Button onClick={() => execute(fetchData)}>Start</Button>;
    case 'loading':
      return <Spinner />;
    case 'success':
      return <div>Data: {state.data}</div>; // TypeScript knows `data` exists
    case 'error':
      return <div>Error: {state.error}</div>; // TypeScript knows `error` exists
  }
}
```

### Type Guards

```typescript
// Type guards for runtime type checking
function isString(value: unknown): value is string {
  return typeof value === 'string';
}

function isCourse(value: unknown): value is Course {
  return (
    typeof value === 'object' &&
    value !== null &&
    'id' in value &&
    'title' in value &&
    'subject' in value &&
    'level' in value
  );
}

function isApiError(error: unknown): error is { message: string; code: number } {
  return (
    typeof error === 'object' &&
    error !== null &&
    'message' in error &&
    'code' in error
  );
}

// Usage in error handling
try {
  const result = await api.createCourse(data);
} catch (error) {
  if (isApiError(error)) {
    console.error(`API Error ${error.code}: ${error.message}`);
  } else {
    console.error('Unknown error:', error);
  }
}
```

## 🎨 Component Type Patterns

### Polymorphic Components

```typescript
// Create polymorphic components with proper typing
type AsProp<C extends React.ElementType> = {
  as?: C;
};

type PropsToOmit<C extends React.ElementType, P> = keyof (AsProp<C> & P);

type PolymorphicComponentProp<
  C extends React.ElementType,
  Props = {}
> = React.PropsWithChildren<Props & AsProp<C>> &
  Omit<React.ComponentPropsWithoutRef<C>, PropsToOmit<C, Props>>;

interface TextProps {
  color?: 'primary' | 'secondary' | 'muted';
  size?: 'sm' | 'md' | 'lg';
}

function Text<C extends React.ElementType = 'span'>({
  as,
  color = 'primary',
  size = 'md',
  children,
  className,
  ...props
}: PolymorphicComponentProp<C, TextProps>) {
  const Component = as || 'span';
  
  return (
    <Component
      className={cn(
        'text-base',
        {
          'text-primary': color === 'primary',
          'text-secondary': color === 'secondary',
          'text-muted-foreground': color === 'muted',
          'text-sm': size === 'sm',
          'text-base': size === 'md',
          'text-lg': size === 'lg',
        },
        className
      )}
      {...props}
    >
      {children}
    </Component>
  );
}

// Usage - component adapts to the 'as' prop
<Text>Default span</Text>
<Text as="h1">Heading</Text>
<Text as="p" color="muted">Paragraph</Text>
<Text as={Link} href="/courses">Link</Text>
```

```typescript
const Input = React.forwardRef<HTMLInputElement, InputProps>(

            "flex h-10 w-full rounded-md border border-input bg-background px-3 py-2",
            error && "border-destructive",
            className
          )}
          ref={ref}
          {...props}
        />
        {error && <p className="text-sm text-destructive">{error}</p>}
      </div>
    );
  }
);
```

### Type Testing with TypeScript

```typescript
// Test types at compile time
type Expect<T extends true> = T;
type Equal<X, Y> = (<T>() => T extends X ? 1 : 2) extends <T>() => T extends Y ? 1 : 2
  ? true
  : false;

// Test utility types
type TestOptional = Optional<User, 'id'>;
type _TestOptional = Expect<Equal<TestOptional, Omit<User, 'id'> & Partial<Pick<User, 'id'>>>>;

// Test component props
type TestButtonProps = React.ComponentProps<typeof Button>;
type _TestButtonVariant = Expect<Equal<
  TestButtonProps['variant'],
  'default' | 'destructive' | 'outline' | 'secondary' | 'ghost' | 'link' | undefined
>>;
```

## 🎯 Best Practices

1. **Strict Configuration**: Use strict TypeScript settings for maximum type safety
2. **Type-First Design**: Define types before implementing components/functions
3. **Generic Constraints**: Use generic constraints to limit type parameters
4. **Discriminated Unions**: Prefer discriminated unions over loose object types
5. **Type Guards**: Implement type guards for runtime type checking
6. **Branded Types**: Use branded types to prevent mixing similar primitive types
7. **Utility Types**: Leverage TypeScript's built-in and custom utility types
8. **Component Typing**: Properly type React components with forwardRef and polymorphism

TypeScript provides the foundation for Edukai's type-safe, maintainable, and robust codebase.
